### 실습 2: 메트릭 수집 파이프라인 테스트

```bash
#!/bin/bash
echo "=== 메트릭 수집 파이프라인 테스트 ==="

# Prometheus 포트 포워딩 (백그라운드)
kubectl port-forward -n monitoring svc/prometheus-k8s 9090:9090 &
PROM_PID=$!
sleep 3

# 1. Sample App 상태 확인
echo "[1/5] Sample App 확인..."
kubectl get pods -l app=sample-app -n default
if [ $? -ne 0 ]; then
    echo "⚠️ Sample App이 없습니다. 배포해주세요."
fi

# 2. ServiceMonitor 확인
echo "[2/5] ServiceMonitor 확인..."
kubectl get servicemonitor sample-app -n monitoring 2>/dev/null
if [ $? -eq 0 ]; then
    echo "✅ ServiceMonitor 존재"
else
    echo "⚠️ ServiceMonitor가 없습니다."
fi

# 3. Prometheus Target 확인
echo "[3/5] Prometheus Target 확인..."
TARGET_RESULT=$(curl -s http://localhost:9090/api/v1/targets | jq -r '.data.activeTargets[] | select(.labels.job=="sample-app") | .health')
if [ "$TARGET_RESULT" = "up" ]; then
    echo "✅ Target이 UP 상태입니다."
else
    echo "⚠️ Target 상태: $TARGET_RESULT"
fi

# 4. Recording Rules 메트릭 확인
echo "[4/5] Recording Rules 메트릭 확인..."
RECORDING_RESULT=$(curl -s "http://localhost:9090/api/v1/query?query=sample_app:http_requests_total:rate5m" | jq -r '.status')
if [ "$RECORDING_RESULT" = "success" ]; then
    echo "✅ Recording Rules 메트릭 정상"
else
    echo "⚠️ Recording Rules 메트릭 조회 실패"
fi

# 5. 메트릭 값 확인
echo "[5/5] 메트릭 값 확인..."
curl -s "http://localhost:9090/api/v1/query?query=up{job='sample-app'}" | jq '.data.result'

# 정리
kill $PROM_PID 2>/dev/null

echo "=== 테스트 완료 ==="
```

### 실습 3: Grafana 대시보드 테스트

```bash
#!/bin/bash
echo "=== Grafana 대시보드 테스트 ==="

# Grafana 포트 포워딩
kubectl port-forward -n monitoring svc/grafana 3000:3000 &
GRAFANA_PID=$!
sleep 3

# Grafana 접속 정보
GRAFANA_USER="admin"
GRAFANA_PASS=$(kubectl get secret grafana-admin -n monitoring -o jsonpath='{.data.admin-password}' | base64 -d 2>/dev/null || echo "admin123")

# 1. 데이터소스 확인
echo "[1/3] 데이터소스 확인..."
DATASOURCES=$(curl -s -u $GRAFANA_USER:$GRAFANA_PASS http://localhost:3000/api/datasources)
echo "데이터소스 목록:"
echo $DATASOURCES | jq -r '.[].name'

# 2. 대시보드 목록 확인
echo "[2/3] 대시보드 목록..."
DASHBOARDS=$(curl -s -u $GRAFANA_USER:$GRAFANA_PASS http://localhost:3000/api/search?type=dash-db)
echo "대시보드 수: $(echo $DASHBOARDS | jq length)"
echo $DASHBOARDS | jq -r '.[].title'

# 3. 대시보드 로딩 테스트
echo "[3/3] 대시보드 로딩 테스트..."
DASHBOARD_UID=$(echo $DASHBOARDS | jq -r '.[0].uid')
if [ -n "$DASHBOARD_UID" ] && [ "$DASHBOARD_UID" != "null" ]; then
    START_TIME=$(date +%s%3N)
    curl -s -o /dev/null -u $GRAFANA_USER:$GRAFANA_PASS "http://localhost:3000/api/dashboards/uid/$DASHBOARD_UID"
    END_TIME=$(date +%s%3N)
    LOAD_TIME=$((END_TIME - START_TIME))
    echo "✅ 대시보드 로딩 시간: ${LOAD_TIME}ms"
else
    echo "⚠️ 대시보드가 없습니다."
fi

# 정리
kill $GRAFANA_PID 2>/dev/null

echo "=== 테스트 완료 ==="
```

## 🛠️ Part 3: 부하 테스트 (1시간)

### 실습 4: 메트릭 수집 부하 테스트

```bash
#!/bin/bash
echo "=== 메트릭 수집 부하 테스트 ==="

# Sample App 포트 포워딩
kubectl port-forward svc/sample-app 8080:8080 -n default &
APP_PID=$!
sleep 2

# 부하 생성 (30초간 초당 50 요청)
echo "부하 생성 중 (30초간)..."
for i in $(seq 1 1500); do
    curl -s http://localhost:8080/ > /dev/null &
    
    # 50개마다 잠시 대기 (초당 50개 유지)
    if [ $((i % 50)) -eq 0 ]; then
        sleep 1
        echo "요청 $i/1500 완료"
    fi
done
wait

echo "부하 생성 완료. Prometheus에서 메트릭 확인..."

# Prometheus에서 메트릭 확인
kubectl port-forward -n monitoring svc/prometheus-k8s 9090:9090 &
PROM_PID=$!
sleep 2

# 요청률 확인
echo "요청률 확인:"
curl -s "http://localhost:9090/api/v1/query?query=rate(http_requests_total{job='sample-app'}[1m])" | jq '.data.result[0].value[1]'

# 정리
kill $APP_PID $PROM_PID 2>/dev/null

echo "=== 부하 테스트 완료 ==="
```

### 실습 5: Prometheus 쿼리 성능 테스트

```bash
#!/bin/bash
echo "=== Prometheus 쿼리 성능 테스트 ==="

kubectl port-forward -n monitoring svc/prometheus-k8s 9090:9090 &
sleep 2

# 쿼리 목록
declare -a QUERIES=(
    "up"
    "rate(http_requests_total[5m])"
    "histogram_quantile(0.95, rate(http_request_duration_seconds_bucket[5m]))"
    "sum(rate(container_cpu_usage_seconds_total[5m])) by (pod)"
)

echo "쿼리 성능 측정:"
for QUERY in "${QUERIES[@]}"; do
    START=$(date +%s%3N)
    curl -s -o /dev/null "http://localhost:9090/api/v1/query?query=$(echo $QUERY | jq -sRr @uri)"
    END=$(date +%s%3N)
    DURATION=$((END - START))
    echo "  $QUERY: ${DURATION}ms"
done

pkill -f "port-forward.*9090"
echo "=== 테스트 완료 ==="
```

### 실습 6: 에러율 상승 시뮬레이션

```bash
#!/bin/bash
echo "=== 에러율 상승 시뮬레이션 ==="

# 에러 생성 Pod 실행
kubectl run error-generator --image=curlimages/curl --restart=Never -- \
  sh -c "while true; do curl -s http://sample-app:8080/error || true; sleep 0.05; done"

echo "Prometheus UI에서 Alerts 탭을 확인하세요."

# 5분 대기
sleep 300

# Alert 상태 확인
kubectl port-forward -n monitoring svc/prometheus-k8s 9090:9090 &
sleep 2

echo "현재 Firing 중인 Alert:"
curl -s http://localhost:9090/api/v1/alerts | jq '.data.alerts[] | select(.state=="firing") | {alertname: .labels.alertname, state: .state}'

# 정리
kubectl delete pod error-generator --ignore-not-found

pkill -f "port-forward.*9090"
echo "=== 시뮬레이션 완료 ==="
```

### 실습 7: Pod 다운 시뮬레이션

```bash
#!/bin/bash
echo "=== Pod 다운 시뮬레이션 ==="

# 현재 Pod 수 확인
CURRENT_REPLICAS=$(kubectl get deployment sample-app -n default -o jsonpath='{.spec.replicas}')
echo "현재 replicas: $CURRENT_REPLICAS"

# Pod 스케일 다운
echo "Sample App 스케일 다운..."
kubectl scale deployment sample-app --replicas=0 -n default

echo "2분 대기 후 Alert 확인..."
sleep 120

# Alert 상태 확인
kubectl port-forward -n monitoring svc/prometheus-k8s 9090:9090 &
sleep 2

echo "현재 Alert 상태:"
curl -s http://localhost:9090/api/v1/alerts | jq '.data.alerts[] | select(.labels.alertname | contains("Down") or contains("Absent"))'

# 복구
echo "Sample App 복구 중..."
kubectl scale deployment sample-app --replicas=$CURRENT_REPLICAS -n default

echo "복구 완료. 잠시 후 Alert가 해제됩니다."

pkill -f "port-forward.*9090"
echo "=== 시뮬레이션 완료 ==="
```

## 💡 면접 대비 핵심 포인트

### Q1: 모니터링 시스템을 어떻게 테스트하나요?
**A**: E2E 테스트로 메트릭 수집부터 알림까지 전체 파이프라인을 검증하고, 장애 시뮬레이션으로 Alert가 제대로 발생하는지 확인합니다.

### Q2: Alert 테스트는 어떻게 하나요?
**A**: 의도적으로 에러를 발생시키거나 Pod를 다운시켜 Alert가 발생하는지 확인하고, 복구 후 Alert가 해제되는지 확인합니다.

### Q3: 부하 테스트에서 무엇을 측정하나요?
**A**: 메트릭 수집 지연, 쿼리 응답 시간, 대시보드 로딩 시간 등을 측정하여 대량 트래픽에서도 모니터링이 정상 동작하는지 확인합니다.
