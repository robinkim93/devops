## 🛠️ Part 2: GoCD Kubernetes 설치

### Helm Chart 설치

```bash
# Helm 레포 추가
helm repo add gocd https://gocd.github.io/helm-chart
helm repo update

# 네임스페이스 생성
kubectl create namespace gocd

# values.yaml 작성
cat << 'EOF' > gocd-values.yaml
server:
  image:
    repository: gocd/gocd-server
    tag: v23.5.0
  
  persistence:
    enabled: true
    storageClass: gp3
    size: 10Gi
  
  ingress:
    enabled: true
    ingressClassName: nginx
    hosts:
      - host: gocd.example.com
        paths:
          - path: /
            pathType: Prefix
    tls:
      - secretName: gocd-tls
        hosts:
          - gocd.example.com

  # 초기 Pipeline as Code 설정
  preconfigureRecreatingAgent:
    enable: true

agent:
  # Static Agent는 비활성화 (Elastic Agent 사용)
  replicaCount: 0
EOF

# 설치
helm install gocd gocd/gocd \
  -n gocd \
  -f gocd-values.yaml
```

### 초기 접속

```bash
# 서비스 확인
kubectl get pods -n gocd
kubectl get svc -n gocd

# Port Forward (Ingress 없을 때)
kubectl port-forward svc/gocd-server -n gocd 8153:8153

# 초기 비밀번호 확인
kubectl get secret gocd-server -n gocd -o jsonpath='{.data.password}' | base64 -d
```

## 🛠️ Part 3: Kubernetes Elastic Agent 설정

### Cluster Profile 생성

```yaml
# GoCD UI: Admin → Elastic Agent Configurations → Add Cluster Profile

# 또는 API 호출
curl -X POST "https://gocd.example.com/go/api/admin/elastic/cluster_profiles" \
  -H "Accept: application/vnd.go.cd.v2+json" \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer ${GOCD_TOKEN}" \
  -d '{
    "id": "k8s-cluster",
    "plugin_id": "cd.go.contrib.elasticagent.kubernetes",
    "properties": [
      {"key": "kubernetes_cluster_url", "value": "https://kubernetes.default.svc"},
      {"key": "namespace", "value": "gocd"},
      {"key": "security_token", "value": ""},
      {"key": "kubernetes_cluster_ca_cert", "value": ""}
    ]
  }'
```

### Elastic Profile 정의

```yaml
# docker-agent 프로필 (이미지 빌드용)
format_version: 10
elastic_profiles:
  docker-agent:
    cluster_profile_id: k8s-cluster
    configuration:
      pod_spec:
        apiVersion: v1
        kind: Pod
        metadata:
          labels:
            app: gocd-agent
            type: docker
        spec:
          serviceAccountName: gocd-agent
          containers:
          - name: gocd-agent
            image: gocd/gocd-agent-docker-dind:v23.5.0
            securityContext:
              privileged: true
            volumeMounts:
            - name: docker-graph-storage
              mountPath: /var/lib/docker
            resources:
              requests:
                cpu: "500m"
                memory: "1Gi"
              limits:
                cpu: "2"
                memory: "4Gi"
          volumes:
          - name: docker-graph-storage
            emptyDir: {}

  # k8s-agent 프로필 (K8s 배포용)
  k8s-agent:
    cluster_profile_id: k8s-cluster
    configuration:
      pod_spec:
        apiVersion: v1
        kind: Pod
        metadata:
          labels:
            app: gocd-agent
            type: k8s
        spec:
          serviceAccountName: gocd-deployer
          containers:
          - name: gocd-agent
            image: gocd/gocd-agent-alpine-3.18:v23.5.0
            command:
            - /docker-entrypoint.sh
            volumeMounts:
            - name: kubeconfig
              mountPath: /root/.kube
              readOnly: true
            resources:
              requests:
                cpu: "200m"
                memory: "256Mi"
              limits:
                cpu: "500m"
                memory: "512Mi"
          volumes:
          - name: kubeconfig
            secret:
              secretName: gocd-kubeconfig

  # helm-agent 프로필 (Helm 배포용)
  helm-agent:
    cluster_profile_id: k8s-cluster
    configuration:
      pod_spec:
        apiVersion: v1
        kind: Pod
        spec:
          serviceAccountName: gocd-deployer
          containers:
          - name: gocd-agent
            image: alpine/helm:3.13.3
            command: ["sleep", "infinity"]
            volumeMounts:
            - name: kubeconfig
              mountPath: /root/.kube
          volumes:
          - name: kubeconfig
            secret:
              secretName: gocd-kubeconfig
```

### RBAC 설정

```yaml
# gocd-rbac.yaml
apiVersion: v1
kind: ServiceAccount
metadata:
  name: gocd-deployer
  namespace: gocd
---
apiVersion: rbac.authorization.k8s.io/v1
kind: ClusterRole
metadata:
  name: gocd-deployer
rules:
- apiGroups: ["", "apps", "extensions"]
  resources: ["deployments", "services", "pods", "configmaps", "secrets"]
  verbs: ["get", "list", "watch", "create", "update", "patch", "delete"]
- apiGroups: ["networking.k8s.io"]
  resources: ["ingresses"]
  verbs: ["get", "list", "watch", "create", "update", "patch", "delete"]
---
apiVersion: rbac.authorization.k8s.io/v1
kind: ClusterRoleBinding
metadata:
  name: gocd-deployer
roleRef:
  apiGroup: rbac.authorization.k8s.io
  kind: ClusterRole
  name: gocd-deployer
subjects:
- kind: ServiceAccount
  name: gocd-deployer
  namespace: gocd
```

```bash
kubectl apply -f gocd-rbac.yaml
```

## 🛠️ Part 4: Kubernetes 배포 파이프라인

### kubectl 배포

```yaml
format_version: 10
pipelines:
  k8s-deploy:
    group: tossplace
    
    materials:
      infra:
        git: https://github.com/tossplace/k8s-manifests.git
        branch: main
    
    stages:
    - deploy-staging:
        approval:
          type: success
        environment_variables:
          NAMESPACE: staging
        jobs:
          deploy:
            elastic_profile_id: k8s-agent
            tasks:
            # 매니페스트 적용
            - exec:
                command: kubectl
                arguments:
                - apply
                - -f
                - manifests/
                - -n
                - ${NAMESPACE}
            
            # Rollout 대기
            - exec:
                command: kubectl
                arguments:
                - rollout
                - status
                - deployment/myapp
                - -n
                - ${NAMESPACE}
                - --timeout=300s
            
            # 상태 확인
            - exec:
                command: kubectl
                arguments:
                - get
                - pods
                - -n
                - ${NAMESPACE}
                - -l
                - app=myapp
```

### Helm 배포

```yaml
format_version: 10
pipelines:
  helm-deploy:
    group: tossplace
    
    materials:
      chart:
        git: https://github.com/tossplace/helm-charts.git
        branch: main
    
    environment_variables:
      RELEASE_NAME: myapp
      CHART_PATH: charts/myapp
    
    stages:
    - deploy-staging:
        environment_variables:
          NAMESPACE: staging
          VALUES_FILE: values-staging.yaml
        jobs:
          helm-upgrade:
            elastic_profile_id: helm-agent
            tasks:
            - exec:
                command: helm
                arguments:
                - dependency
                - update
                - ${CHART_PATH}
            
            - exec:
                command: helm
                arguments:
                - upgrade
                - --install
                - ${RELEASE_NAME}
                - ${CHART_PATH}
                - -n
                - ${NAMESPACE}
                - -f
                - ${CHART_PATH}/${VALUES_FILE}
                - --set
                - image.tag=${GO_PIPELINE_LABEL}
                - --wait
                - --timeout
                - 5m
    
    - deploy-production:
        approval:
          type: manual
        environment_variables:
          NAMESPACE: production
          VALUES_FILE: values-production.yaml
        jobs:
          helm-upgrade:
            elastic_profile_id: helm-agent
            tasks:
            - exec:
                command: helm
                arguments:
                - upgrade
                - --install
                - ${RELEASE_NAME}
                - ${CHART_PATH}
                - -n
                - ${NAMESPACE}
                - -f
                - ${CHART_PATH}/${VALUES_FILE}
                - --set
                - image.tag=${GO_PIPELINE_LABEL}
                - --wait
                - --timeout
                - 10m
```

## 🔍 트러블슈팅

### 문제 1: Elastic Agent 생성 안됨

```bash
# GoCD 로그 확인
kubectl logs -n gocd gocd-server-0 -f

# 에러: ServiceAccount 권한 부족
# 해결: RBAC 설정 확인
kubectl auth can-i create pods --as=system:serviceaccount:gocd:gocd-deployer -n gocd
```

### 문제 2: kubectl 명령 실패

```bash
# kubeconfig Secret 확인
kubectl get secret gocd-kubeconfig -n gocd

# kubeconfig 생성
kubectl create secret generic gocd-kubeconfig \
  -n gocd \
  --from-file=config=$HOME/.kube/config
```
