## 🛠️ Part 1: kube-prometheus-stack 설치 (1시간)

### Helm 설치

```bash
# Helm repo 추가
helm repo add prometheus-community https://prometheus-community.github.io/helm-charts
helm repo update

# monitoring 네임스페이스 생성
kubectl create namespace monitoring

# values 파일 생성
cat > monitoring/prometheus-values.yaml << 'EOF'
# kube-prometheus-stack Helm values
# 토스플레이스 인프라 클론

prometheus:
  prometheusSpec:
    # 리소스 설정
    resources:
      requests:
        memory: 2Gi
        cpu: 500m
      limits:
        memory: 4Gi
        cpu: 2000m
    
    # 보존 기간 (Thanos로 장기 저장)
    retention: 24h
    retentionSize: "10GB"
    
    # Thanos Sidecar 활성화
    thanos:
      image: quay.io/thanos/thanos:v0.32.5
      version: v0.32.5
      objectStorageConfig:
        key: thanos.yaml
        name: thanos-objstore-config
    
    # ServiceMonitor 선택
    serviceMonitorSelectorNilUsesHelmValues: false
    podMonitorSelectorNilUsesHelmValues: false
    ruleSelectorNilUsesHelmValues: false
    
    # 추가 스크랩 설정
    additionalScrapeConfigs:
    - job_name: 'istio-mesh'
      kubernetes_sd_configs:
      - role: endpoints
        namespaces:
          names:
          - istio-system
      relabel_configs:
      - source_labels: [__meta_kubernetes_service_name]
        regex: istiod
        action: keep
  
  # 외부 레이블 (Thanos 쿼리용)
  thanosRuler:
    enabled: false

grafana:
  enabled: true
  adminPassword: "admin-password-change-me"
  
  # 영구 저장소
  persistence:
    enabled: true
    size: 10Gi
  
  # 사이드카 (대시보드 자동 로드)
  sidecar:
    dashboards:
      enabled: true
      searchNamespace: ALL
    datasources:
      enabled: true
  
  # 데이터소스 설정
  additionalDataSources:
  - name: Thanos
    type: prometheus
    url: http://thanos-query:9090
    access: proxy
    isDefault: true
  - name: Loki
    type: loki
    url: http://loki:3100
    access: proxy

alertmanager:
  alertmanagerSpec:
    resources:
      requests:
        memory: 256Mi
        cpu: 100m
  
  config:
    global:
      resolve_timeout: 5m
      slack_api_url: 'https://hooks.slack.com/services/YOUR/SLACK/WEBHOOK'
    
    route:
      group_by: ['alertname', 'namespace']
      group_wait: 30s
      group_interval: 5m
      repeat_interval: 4h
      receiver: 'slack-notifications'
      routes:
      - match:
          severity: critical
        receiver: 'slack-critical'
        continue: true
      - match:
          severity: warning
        receiver: 'slack-notifications'
    
    receivers:
    - name: 'slack-notifications'
      slack_configs:
      - channel: '#alerts'
        send_resolved: true
        title: '{{ template "slack.default.title" . }}'
        text: '{{ template "slack.default.text" . }}'
    
    - name: 'slack-critical'
      slack_configs:
      - channel: '#alerts-critical'
        send_resolved: true
        title: '🚨 CRITICAL: {{ template "slack.default.title" . }}'
        text: '{{ template "slack.default.text" . }}'

# Node Exporter
nodeExporter:
  enabled: true

# Kube State Metrics
kubeStateMetrics:
  enabled: true

# Prometheus Operator
prometheusOperator:
  enabled: true
  resources:
    requests:
      memory: 256Mi
      cpu: 100m
EOF

# 설치
helm install prometheus prometheus-community/kube-prometheus-stack \
  -n monitoring \
  -f monitoring/prometheus-values.yaml
```

## 🛠️ Part 2: Thanos 설정 (1시간 30분)

### S3 Object Store 설정

```bash
# Thanos Object Store 시크릿
cat > monitoring/thanos-objstore-secret.yaml << 'EOF'
apiVersion: v1
kind: Secret
metadata:
  name: thanos-objstore-config
  namespace: monitoring
type: Opaque
stringData:
  thanos.yaml: |
    type: S3
    config:
      bucket: tossplace-clone-thanos-metrics
      endpoint: s3.ap-northeast-2.amazonaws.com
      region: ap-northeast-2
EOF

kubectl apply -f monitoring/thanos-objstore-secret.yaml
```

### Thanos 컴포넌트 설치

```bash
# Thanos Helm 설치
helm repo add bitnami https://charts.bitnami.com/bitnami

cat > monitoring/thanos-values.yaml << 'EOF'
# Thanos Helm Values

objstoreConfig: |-
  type: S3
  config:
    bucket: tossplace-clone-thanos-metrics
    endpoint: s3.ap-northeast-2.amazonaws.com
    region: ap-northeast-2

query:
  enabled: true
  replicaCount: 2
  
  stores:
  - "dnssrv+_grpc._tcp.prometheus-prometheus-kube-prometheus-thanos-discovery.monitoring.svc.cluster.local"
  - "dnssrv+_grpc._tcp.thanos-storegateway.monitoring.svc.cluster.local"
  
  resources:
    requests:
      memory: 256Mi
      cpu: 100m

queryFrontend:
  enabled: true
  replicaCount: 2
  
  config: |-
    type: IN-MEMORY
    config:
      max_size: 512MB
      max_size_items: 1000
      validity: 6h

storegateway:
  enabled: true
  replicaCount: 2
  
  persistence:
    enabled: true
    size: 20Gi
  
  resources:
    requests:
      memory: 512Mi
      cpu: 200m

compactor:
  enabled: true
  
  retentionResolutionRaw: 30d
  retentionResolution5m: 90d
  retentionResolution1h: 1y
  
  persistence:
    enabled: true
    size: 50Gi
  
  resources:
    requests:
      memory: 512Mi
      cpu: 200m

ruler:
  enabled: false

receive:
  enabled: false

# 메트릭 수집
metrics:
  enabled: true
  serviceMonitor:
    enabled: true
EOF

helm install thanos bitnami/thanos \
  -n monitoring \
  -f monitoring/thanos-values.yaml
```

## 🛠️ Part 4: AlertManager 규칙 (30분)

### PrometheusRule 생성

```bash
# monitoring/prometheus-rules.yaml
cat > monitoring/prometheus-rules.yaml << 'EOF'
apiVersion: monitoring.coreos.com/v1
kind: PrometheusRule
metadata:
  name: tossplace-alerts
  namespace: monitoring
  labels:
    release: prometheus
spec:
  groups:
  - name: kubernetes-apps
    rules:
    - alert: KubePodCrashLooping
      expr: rate(kube_pod_container_status_restarts_total[15m]) > 0
      for: 1h
      labels:
        severity: critical
      annotations:
        summary: "Pod {{ $labels.namespace }}/{{ $labels.pod }} 재시작 반복"
        description: "Pod {{ $labels.namespace }}/{{ $labels.pod }}가 지난 1시간 동안 재시작을 반복하고 있습니다."
    
    - alert: KubePodNotReady
      expr: sum by (namespace, pod) (kube_pod_status_phase{phase=~"Pending|Unknown"}) > 0
      for: 15m
      labels:
        severity: critical
      annotations:
        summary: "Pod {{ $labels.namespace }}/{{ $labels.pod }} Not Ready"
        description: "Pod {{ $labels.namespace }}/{{ $labels.pod }}가 15분 이상 Ready 상태가 아닙니다."
  
  - name: kubernetes-resources
    rules:
    - alert: KubeNodeNotReady
      expr: kube_node_status_condition{condition="Ready",status="true"} == 0
      for: 5m
      labels:
        severity: critical
      annotations:
        summary: "노드 {{ $labels.node }} Not Ready"
        description: "노드 {{ $labels.node }}가 5분 이상 Ready 상태가 아닙니다."
    
    - alert: HighCPUUsage
      expr: 100 - (avg by(instance) (rate(node_cpu_seconds_total{mode="idle"}[5m])) * 100) > 80
      for: 5m
      labels:
        severity: warning
      annotations:
        summary: "노드 {{ $labels.instance }} CPU 사용률 높음"
        description: "노드 {{ $labels.instance }}의 CPU 사용률이 80%를 초과했습니다. 현재: {{ $value }}%"
    
    - alert: HighMemoryUsage
      expr: (1 - (node_memory_MemAvailable_bytes / node_memory_MemTotal_bytes)) * 100 > 85
      for: 5m
      labels:
        severity: warning
      annotations:
        summary: "노드 {{ $labels.instance }} 메모리 사용률 높음"
        description: "노드 {{ $labels.instance }}의 메모리 사용률이 85%를 초과했습니다. 현재: {{ $value }}%"
  
  - name: istio
    rules:
    - alert: HighRequestLatency
      expr: histogram_quantile(0.99, sum(rate(istio_request_duration_milliseconds_bucket[5m])) by (le, destination_service)) > 2000
      for: 5m
      labels:
        severity: warning
      annotations:
        summary: "서비스 {{ $labels.destination_service }} P99 지연 높음"
        description: "{{ $labels.destination_service }}의 P99 응답 시간이 2초를 초과했습니다."
    
    - alert: High5xxErrorRate
      expr: sum(rate(istio_requests_total{response_code=~"5.."}[5m])) by (destination_service) / sum(rate(istio_requests_total[5m])) by (destination_service) > 0.05
      for: 5m
      labels:
        severity: critical
      annotations:
        summary: "서비스 {{ $labels.destination_service }} 5xx 에러율 높음"
        description: "{{ $labels.destination_service }}의 5xx 에러율이 5%를 초과했습니다."
EOF

kubectl apply -f monitoring/prometheus-rules.yaml
```
