## 🛠️ Part 1: Istio 설치 (1시간)

### 사전 준비

```bash
# 1. EKS 클러스터 접속 확인
kubectl get nodes

# 2. istioctl 설치
curl -L https://istio.io/downloadIstio | ISTIO_VERSION=1.20.0 sh -
export PATH=$PWD/istio-1.20.0/bin:$PATH

# 3. 사전 검증
istioctl x precheck
```

### Istio Production Profile 설치

```bash
# istio/install.yaml
cat > istio/install.yaml << 'EOF'
apiVersion: install.istio.io/v1alpha1
kind: IstioOperator
metadata:
  name: istio-controlplane
  namespace: istio-system
spec:
  profile: default
  
  # 메시 전역 설정
  meshConfig:
    accessLogFile: /dev/stdout
    accessLogFormat: |
      [%START_TIME%] "%REQ(:METHOD)% %REQ(X-ENVOY-ORIGINAL-PATH?:PATH)% %PROTOCOL%"
      %RESPONSE_CODE% %RESPONSE_FLAGS% %BYTES_RECEIVED% %BYTES_SENT%
      %DURATION% %RESP(X-ENVOY-UPSTREAM-SERVICE-TIME)%
      "%REQ(X-FORWARDED-FOR)%" "%REQ(USER-AGENT)%" "%REQ(X-REQUEST-ID)%"
      "%REQ(:AUTHORITY)%" "%UPSTREAM_HOST%"
    
    # 기본 mTLS 활성화
    defaultConfig:
      proxyMetadata:
        ISTIO_META_DNS_CAPTURE: "true"
        ISTIO_META_DNS_AUTO_ALLOCATE: "true"
    
    # 분산 추적 (Jaeger)
    defaultProviders:
      tracing:
      - zipkin
    
    enableTracing: true
    
    # Outbound 트래픽 정책
    outboundTrafficPolicy:
      mode: REGISTRY_ONLY
  
  # 컴포넌트 설정
  components:
    # Ingress Gateway - Production 설정
    ingressGateways:
    - name: istio-ingressgateway
      enabled: true
      k8s:
        hpaSpec:
          minReplicas: 2
          maxReplicas: 10
          metrics:
          - type: Resource
            resource:
              name: cpu
              target:
                type: Utilization
                averageUtilization: 80
        resources:
          requests:
            cpu: 100m
            memory: 128Mi
          limits:
            cpu: 2000m
            memory: 1024Mi
        service:
          type: LoadBalancer
          ports:
          - port: 80
            targetPort: 8080
            name: http2
          - port: 443
            targetPort: 8443
            name: https
    
    # Egress Gateway
    egressGateways:
    - name: istio-egressgateway
      enabled: true
      k8s:
        hpaSpec:
          minReplicas: 1
          maxReplicas: 5
    
    # istiod (Control Plane)
    pilot:
      enabled: true
      k8s:
        hpaSpec:
          minReplicas: 2
          maxReplicas: 5
        resources:
          requests:
            cpu: 100m
            memory: 256Mi
  
  # 프록시 전역 설정
  values:
    global:
      proxy:
        resources:
          requests:
            cpu: 50m
            memory: 64Mi
          limits:
            cpu: 500m
            memory: 256Mi
        
        # 로깅 설정
        logLevel: warning
        componentLogLevel: "misc:error"
    
    # Pilot 설정
    pilot:
      traceSampling: 100.0  # 프로덕션에서는 1.0 권장
EOF

# Istio 설치
istioctl install -f istio/install.yaml -y

# 설치 확인
kubectl get pods -n istio-system
kubectl get svc -n istio-system
```

**예상 출력**:
```
NAME                                    READY   STATUS    RESTARTS   AGE
istio-egressgateway-xxx                 1/1     Running   0          2m
istio-ingressgateway-xxx                1/1     Running   0          2m
istiod-xxx                              1/1     Running   0          2m
```

## 🛠️ Part 2: Gateway 및 트래픽 관리 (1시간 30분)

### Istio Gateway

```bash
# istio/gateway.yaml
cat > istio/gateway.yaml << 'EOF'
apiVersion: networking.istio.io/v1beta1
kind: Gateway
metadata:
  name: tossplace-gateway
  namespace: tossplace-app
spec:
  selector:
    istio: ingressgateway
  servers:
  - port:
      number: 80
      name: http
      protocol: HTTP
    hosts:
    - "*.tossplace.local"
    - "tossplace.local"
    # HTTPS 리다이렉트
    tls:
      httpsRedirect: false  # 테스트 환경
  
  - port:
      number: 443
      name: https
      protocol: HTTPS
    hosts:
    - "*.tossplace.local"
    tls:
      mode: SIMPLE
      credentialName: tossplace-tls  # TLS 인증서 시크릿
EOF

kubectl apply -f istio/gateway.yaml
```

### VirtualService - 카나리 배포 예시

```bash
# istio/virtualservice.yaml
cat > istio/virtualservice.yaml << 'EOF'
apiVersion: networking.istio.io/v1beta1
kind: VirtualService
metadata:
  name: frontend
  namespace: tossplace-app
spec:
  hosts:
  - frontend
  - "tossplace.local"
  gateways:
  - tossplace-gateway
  - mesh
  http:
  - match:
    - uri:
        prefix: /
    route:
    - destination:
        host: frontend
        subset: stable
        port:
          number: 80
      weight: 90
    - destination:
        host: frontend
        subset: canary
        port:
          number: 80
      weight: 10
    
    # 타임아웃 설정
    timeout: 10s
    
    # 재시도 정책
    retries:
      attempts: 3
      perTryTimeout: 3s
      retryOn: gateway-error,connect-failure,refused-stream

---
apiVersion: networking.istio.io/v1beta1
kind: VirtualService
metadata:
  name: backend
  namespace: tossplace-app
spec:
  hosts:
  - backend
  http:
  - route:
    - destination:
        host: backend
        port:
          number: 8080
    timeout: 15s
    retries:
      attempts: 3
      perTryTimeout: 5s
EOF

kubectl apply -f istio/virtualservice.yaml
```

### DestinationRule - Circuit Breaker

```bash
# istio/destination-rule.yaml
cat > istio/destination-rule.yaml << 'EOF'
apiVersion: networking.istio.io/v1beta1
kind: DestinationRule
metadata:
  name: frontend
  namespace: tossplace-app
spec:
  host: frontend
  trafficPolicy:
    connectionPool:
      tcp:
        maxConnections: 100
      http:
        h2UpgradePolicy: UPGRADE
        http1MaxPendingRequests: 100
        http2MaxRequests: 1000
        maxRequestsPerConnection: 10
    
    outlierDetection:
      consecutive5xxErrors: 5
      interval: 10s
      baseEjectionTime: 30s
      maxEjectionPercent: 50
      minHealthPercent: 30
  
  subsets:
  - name: stable
    labels:
      version: stable
  - name: canary
    labels:
      version: canary

---
apiVersion: networking.istio.io/v1beta1
kind: DestinationRule
metadata:
  name: backend
  namespace: tossplace-app
spec:
  host: backend
  trafficPolicy:
    connectionPool:
      tcp:
        maxConnections: 100
      http:
        h2UpgradePolicy: UPGRADE
        http1MaxPendingRequests: 50
        http2MaxRequests: 500
    
    outlierDetection:
      consecutive5xxErrors: 3
      interval: 5s
      baseEjectionTime: 60s
      maxEjectionPercent: 100
  
  subsets:
  - name: v1
    labels:
      version: v1
EOF

kubectl apply -f istio/destination-rule.yaml
```

## 🛠️ Part 3: mTLS 설정 (1시간)

### PeerAuthentication - STRICT mTLS

```bash
# istio/peer-authentication.yaml
cat > istio/peer-authentication.yaml << 'EOF'
# 메시 전체 STRICT mTLS
apiVersion: security.istio.io/v1beta1
kind: PeerAuthentication
metadata:
  name: default
  namespace: istio-system
spec:
  mtls:
    mode: STRICT

---
# 네임스페이스별 STRICT mTLS
apiVersion: security.istio.io/v1beta1
kind: PeerAuthentication
metadata:
  name: tossplace-mtls
  namespace: tossplace-app
spec:
  mtls:
    mode: STRICT
EOF

kubectl apply -f istio/peer-authentication.yaml
```

### AuthorizationPolicy - 서비스 간 접근 제어

```bash
# istio/authorization-policy.yaml
cat > istio/authorization-policy.yaml << 'EOF'
# 기본 Deny 정책
apiVersion: security.istio.io/v1beta1
kind: AuthorizationPolicy
metadata:
  name: deny-all
  namespace: tossplace-app
spec:
  {}

---
# Frontend -> Backend 허용
apiVersion: security.istio.io/v1beta1
kind: AuthorizationPolicy
metadata:
  name: allow-frontend-to-backend
  namespace: tossplace-app
spec:
  selector:
    matchLabels:
      app: backend
  action: ALLOW
  rules:
  - from:
    - source:
        principals:
        - "cluster.local/ns/tossplace-app/sa/frontend"
    to:
    - operation:
        methods: ["GET", "POST", "PUT", "DELETE"]
        paths: ["/api/*"]

---
# Ingress Gateway -> Frontend 허용
apiVersion: security.istio.io/v1beta1
kind: AuthorizationPolicy
metadata:
  name: allow-ingress-to-frontend
  namespace: tossplace-app
spec:
  selector:
    matchLabels:
      app: frontend
  action: ALLOW
  rules:
  - from:
    - source:
        namespaces: ["istio-system"]
    to:
    - operation:
        methods: ["GET", "POST"]

---
# Backend -> RDS 외부 허용 (Egress)
apiVersion: security.istio.io/v1beta1
kind: AuthorizationPolicy
metadata:
  name: allow-backend-egress
  namespace: tossplace-app
spec:
  selector:
    matchLabels:
      app: backend
  action: ALLOW
  rules:
  - to:
    - operation:
        ports: ["3306", "6379"]
EOF

kubectl apply -f istio/authorization-policy.yaml
```
