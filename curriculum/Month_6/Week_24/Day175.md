## 🛠️ Part 1: ArgoCD 설치 (1시간)

### ArgoCD HA 설치

```bash
# ArgoCD 네임스페이스 생성
kubectl create namespace argocd

# HA 버전 설치 (프로덕션용)
kubectl apply -n argocd -f https://raw.githubusercontent.com/argoproj/argo-cd/v2.9.3/manifests/ha/install.yaml

# 설치 확인
kubectl get pods -n argocd -w
```

### ArgoCD CLI 설치 및 접속

```bash
# CLI 설치 (macOS)
brew install argocd

# 초기 비밀번호 확인
kubectl -n argocd get secret argocd-initial-admin-secret -o jsonpath="{.data.password}" | base64 -d

# 포트 포워딩
kubectl port-forward svc/argocd-server -n argocd 8080:443 &

# 로그인
argocd login localhost:8080 --username admin --password <password> --insecure

# 비밀번호 변경
argocd account update-password
```

### Ingress 설정 (옵션)

```bash
# argocd/ingress.yaml
cat > kubernetes/argocd/ingress.yaml << 'EOF'
apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: argocd-server-ingress
  namespace: argocd
  annotations:
    kubernetes.io/ingress.class: "nginx"
    nginx.ingress.kubernetes.io/ssl-passthrough: "true"
    nginx.ingress.kubernetes.io/backend-protocol: "HTTPS"
spec:
  rules:
  - host: argocd.tossplace.local
    http:
      paths:
      - path: /
        pathType: Prefix
        backend:
          service:
            name: argocd-server
            port:
              number: 443
EOF

kubectl apply -f kubernetes/argocd/ingress.yaml
```

## 🛠️ Part 2: Application 설정 (1시간 30분)

### AppProject 생성

```bash
# kubernetes/argocd/project.yaml
cat > kubernetes/argocd/project.yaml << 'EOF'
apiVersion: argoproj.io/v1alpha1
kind: AppProject
metadata:
  name: tossplace
  namespace: argocd
spec:
  description: "토스플레이스 인프라 클론 프로젝트"
  
  # 소스 레포지토리 제한
  sourceRepos:
  - "https://github.com/YOUR_USERNAME/tossplace-infra-clone"
  - "https://github.com/YOUR_USERNAME/tossplace-app"
  
  # 배포 대상 제한
  destinations:
  - namespace: tossplace-app
    server: https://kubernetes.default.svc
  - namespace: monitoring
    server: https://kubernetes.default.svc
  - namespace: istio-system
    server: https://kubernetes.default.svc
  
  # 클러스터 리소스 허용
  clusterResourceWhitelist:
  - group: ""
    kind: Namespace
  - group: "networking.istio.io"
    kind: "*"
  - group: "security.istio.io"
    kind: "*"
  
  # 네임스페이스 리소스 허용
  namespaceResourceWhitelist:
  - group: "*"
    kind: "*"
  
  # Orphaned 리소스 경고
  orphanedResources:
    warn: true
    ignore:
    - group: ""
      kind: ConfigMap
      name: kube-root-ca.crt
EOF

kubectl apply -f kubernetes/argocd/project.yaml
```

### Application - 샘플 앱 배포

```bash
# kubernetes/argocd/application-frontend.yaml
cat > kubernetes/argocd/application-frontend.yaml << 'EOF'
apiVersion: argoproj.io/v1alpha1
kind: Application
metadata:
  name: frontend
  namespace: argocd
  finalizers:
  - resources-finalizer.argocd.argoproj.io
spec:
  project: tossplace
  
  source:
    repoURL: https://github.com/YOUR_USERNAME/tossplace-infra-clone
    targetRevision: HEAD
    path: kubernetes/overlays/dev
    
    kustomize:
      images:
      - frontend=YOUR_ECR_URL/frontend:latest
  
  destination:
    server: https://kubernetes.default.svc
    namespace: tossplace-app
  
  syncPolicy:
    automated:
      prune: true
      selfHeal: true
      allowEmpty: false
    
    syncOptions:
    - CreateNamespace=true
    - PrunePropagationPolicy=foreground
    - PruneLast=true
    
    retry:
      limit: 5
      backoff:
        duration: 5s
        factor: 2
        maxDuration: 3m
  
  # 헬스 체크 설정
  ignoreDifferences:
  - group: apps
    kind: Deployment
    jsonPointers:
    - /spec/replicas
  
  revisionHistoryLimit: 10
EOF

kubectl apply -f kubernetes/argocd/application-frontend.yaml
```

### ApplicationSet - 멀티 환경 자동 생성

```bash
# kubernetes/argocd/applicationset.yaml
cat > kubernetes/argocd/applicationset.yaml << 'EOF'
apiVersion: argoproj.io/v1alpha1
kind: ApplicationSet
metadata:
  name: tossplace-apps
  namespace: argocd
spec:
  generators:
  - matrix:
      generators:
      # 환경 목록
      - list:
          elements:
          - environment: dev
            namespace: tossplace-dev
          - environment: prod
            namespace: tossplace-prod
      # 앱 목록
      - list:
          elements:
          - app: frontend
            path: kubernetes/apps/frontend
          - app: backend
            path: kubernetes/apps/backend
  
  template:
    metadata:
      name: "{{app}}-{{environment}}"
      namespace: argocd
      labels:
        app: "{{app}}"
        environment: "{{environment}}"
    spec:
      project: tossplace
      
      source:
        repoURL: https://github.com/YOUR_USERNAME/tossplace-infra-clone
        targetRevision: HEAD
        path: "{{path}}/overlays/{{environment}}"
      
      destination:
        server: https://kubernetes.default.svc
        namespace: "{{namespace}}"
      
      syncPolicy:
        automated:
          prune: true
          selfHeal: true
        syncOptions:
        - CreateNamespace=true
EOF

kubectl apply -f kubernetes/argocd/applicationset.yaml
```

## 🛠️ Part 3: Notification 설정 (30분)

### Slack 알림 설정

```bash
# kubernetes/argocd/notification-secret.yaml
cat > kubernetes/argocd/notification-secret.yaml << 'EOF'
apiVersion: v1
kind: Secret
metadata:
  name: argocd-notifications-secret
  namespace: argocd
type: Opaque
stringData:
  slack-token: "xoxb-your-slack-token"
EOF

kubectl apply -f kubernetes/argocd/notification-secret.yaml
```

```bash
# kubernetes/argocd/notification-cm.yaml
cat > kubernetes/argocd/notification-cm.yaml << 'EOF'
apiVersion: v1
kind: ConfigMap
metadata:
  name: argocd-notifications-cm
  namespace: argocd
data:
  service.slack: |
    token: $slack-token
    
  template.app-deployed: |
    message: |
      {{if eq .serviceType "slack"}}:white_check_mark:{{end}} *{{.app.metadata.name}}* 배포 완료!
      환경: {{.app.spec.destination.namespace}}
      커밋: {{.app.status.sync.revision}}
      
  template.app-health-degraded: |
    message: |
      {{if eq .serviceType "slack"}}:warning:{{end}} *{{.app.metadata.name}}* 상태 이상!
      상태: {{.app.status.health.status}}
      환경: {{.app.spec.destination.namespace}}
      
  template.app-sync-failed: |
    message: |
      {{if eq .serviceType "slack"}}:x:{{end}} *{{.app.metadata.name}}* 동기화 실패!
      에러: {{.app.status.operationState.message}}
      
  trigger.on-deployed: |
    - when: app.status.operationState.phase in ['Succeeded'] and app.status.health.status == 'Healthy'
      send: [app-deployed]
      
  trigger.on-health-degraded: |
    - when: app.status.health.status == 'Degraded'
      send: [app-health-degraded]
      
  trigger.on-sync-failed: |
    - when: app.status.operationState.phase in ['Error', 'Failed']
      send: [app-sync-failed]
      
  subscriptions: |
    - recipients:
      - slack:deployments
      triggers:
      - on-deployed
      - on-health-degraded
      - on-sync-failed
EOF

kubectl apply -f kubernetes/argocd/notification-cm.yaml
```

## 🛠️ Part 4: RBAC 설정 (30분)

```bash
# kubernetes/argocd/rbac-cm.yaml
cat > kubernetes/argocd/rbac-cm.yaml << 'EOF'
apiVersion: v1
kind: ConfigMap
metadata:
  name: argocd-rbac-cm
  namespace: argocd
data:
  policy.csv: |
    # 역할 정의
    p, role:readonly, applications, get, */*, allow
    p, role:readonly, clusters, get, *, allow
    p, role:readonly, repositories, get, *, allow
    p, role:readonly, projects, get, *, allow
    
    p, role:developer, applications, get, */*, allow
    p, role:developer, applications, sync, */*, allow
    p, role:developer, applications, action/*, */*, allow
    p, role:developer, logs, get, */*, allow
    p, role:developer, exec, create, */*, allow
    
    p, role:admin, *, *, */*, allow
    
    # 그룹 매핑
    g, dev-team, role:developer
    g, ops-team, role:admin
    g, viewer-team, role:readonly
    
  policy.default: role:readonly
  
  scopes: "[groups]"
EOF

kubectl apply -f kubernetes/argocd/rbac-cm.yaml
```
