### 실습 1: 표준 프로젝트 구조 생성 (20분)

**목표**: 프로덕션 수준 디렉토리 구조 세팅

```bash
# 프로젝트 구조 생성 스크립트
cat > init-project.sh << 'EOF'
#!/bin/bash
set -e

PROJECT_NAME=${1:-"tossplace-infra"}

echo "🚀 Creating project structure: $PROJECT_NAME"

# 디렉토리 생성
mkdir -p $PROJECT_NAME/{modules/{vpc,eks,rds,security-group},environments/{dev,staging,prod},global/{iam,s3}}

# 모듈 기본 파일 생성
for module in vpc eks rds security-group; do
  cat > $PROJECT_NAME/modules/$module/main.tf << 'MODULE_EOF'
# Main module configuration
# TODO: Implement module logic
MODULE_EOF

  cat > $PROJECT_NAME/modules/$module/variables.tf << 'MODULE_EOF'
variable "environment" {
  description = "Environment name (dev, staging, prod)"
  type        = string
}

variable "project_name" {
  description = "Project name for resource naming"
  type        = string
}

variable "tags" {
  description = "Additional tags for resources"
  type        = map(string)
  default     = {}
}
MODULE_EOF

  cat > $PROJECT_NAME/modules/$module/outputs.tf << 'MODULE_EOF'
# Module outputs
# TODO: Define outputs
MODULE_EOF

  cat > $PROJECT_NAME/modules/$module/README.md << MODULE_EOF
# $module Module

## Description
This module creates $module resources.

## Usage
\`\`\`hcl
module "$module" {
  source = "../../modules/$module"
  
  environment  = "dev"
  project_name = "tossplace"
}
\`\`\`

## Inputs
| Name | Description | Type | Required |
|------|-------------|------|----------|
| environment | Environment name | string | yes |
| project_name | Project name | string | yes |
| tags | Additional tags | map(string) | no |

## Outputs
| Name | Description |
|------|-------------|

MODULE_EOF
done

# 환경별 파일 생성
for env in dev staging prod; do
  cat > $PROJECT_NAME/environments/$env/main.tf << MAIN_EOF
# $env Environment Configuration

terraform {
  required_version = ">= 1.5.0"
  
  required_providers {
    aws = {
      source  = "hashicorp/aws"
      version = "~> 5.0"
    }
  }
}

provider "aws" {
  region = var.aws_region
  
  default_tags {
    tags = {
      Environment = "$env"
      Project     = var.project_name
      ManagedBy   = "Terraform"
    }
  }
}

# VPC Module
module "vpc" {
  source = "../../modules/vpc"
  
  environment  = "$env"
  project_name = var.project_name
  tags         = var.additional_tags
}

# EKS Module
module "eks" {
  source = "../../modules/eks"
  
  environment  = "$env"
  project_name = var.project_name
  tags         = var.additional_tags
}
MAIN_EOF

  cat > $PROJECT_NAME/environments/$env/variables.tf << 'VAR_EOF'
variable "aws_region" {
  description = "AWS region"
  type        = string
  default     = "ap-northeast-2"
}

variable "project_name" {
  description = "Project name"
  type        = string
}

variable "additional_tags" {
  description = "Additional tags for all resources"
  type        = map(string)
  default     = {}
}
VAR_EOF

  cat > $PROJECT_NAME/environments/$env/backend.tf << BACKEND_EOF
terraform {
  backend "s3" {
    bucket         = "tossplace-terraform-state"
    key            = "$env/terraform.tfstate"
    region         = "ap-northeast-2"
    encrypt        = true
    dynamodb_table = "terraform-lock"
  }
}
BACKEND_EOF

  cat > $PROJECT_NAME/environments/$env/terraform.tfvars << TFVARS_EOF
project_name = "tossplace"
aws_region   = "ap-northeast-2"

additional_tags = {
  Team      = "Platform"
  CostCenter = "$env"
}
TFVARS_EOF
done

echo "✅ Project structure created successfully!"
echo ""
echo "Next steps:"
echo "  cd $PROJECT_NAME"
echo "  terraform -chdir=environments/dev init"
EOF

chmod +x init-project.sh
./init-project.sh tossplace-infra
```

**예상 출력**:
```
🚀 Creating project structure: tossplace-infra
✅ Project structure created successfully!

Next steps:
  cd tossplace-infra
  terraform -chdir=environments/dev init
```

### 실습 2: TFLint 설정 및 실행 (20분)

**목표**: 코드 품질 자동 검사 구성

```bash
cd tossplace-infra

# TFLint 설정 파일 생성
cat > .tflint.hcl << 'EOF'
# TFLint Configuration

config {
  # Terraform 모듈 경로
  module = true
  
  # 강제 종료 여부
  force = false
}

# AWS Provider 플러그인
plugin "aws" {
  enabled = true
  version = "0.27.0"
  source  = "github.com/terraform-linters/tflint-ruleset-aws"
}

# Terraform 규칙 플러그인
plugin "terraform" {
  enabled = true
  version = "0.5.0"
  source  = "github.com/terraform-linters/tflint-ruleset-terraform"
  preset  = "recommended"
}

# 커스텀 규칙
rule "terraform_naming_convention" {
  enabled = true
  format  = "snake_case"
}

rule "terraform_documented_variables" {
  enabled = true
}

rule "terraform_documented_outputs" {
  enabled = true
}

rule "terraform_typed_variables" {
  enabled = true
}

# AWS 특정 규칙
rule "aws_instance_invalid_type" {
  enabled = true
}

rule "aws_instance_previous_type" {
  enabled = true
}

rule "aws_resource_missing_tags" {
  enabled = true
  tags    = ["Environment", "Project", "ManagedBy"]
}
EOF

# TFLint 플러그인 설치
tflint --init

# TFLint 실행
tflint --recursive

# 예상 출력:
# 4 issue(s) found:
# 
# Warning: Missing required tags: "Environment", "Project", "ManagedBy"
```

### 실습 3: Checkov 보안 스캔 (25분)

```bash
# 취약한 코드 예시 생성 (테스트용)
cat > environments/dev/insecure.tf << 'EOF'
# 보안 취약점이 있는 코드 (테스트용)

# ❌ 문제 1: 퍼블릭 S3 버킷
resource "aws_s3_bucket" "public_bucket" {
  bucket = "my-public-bucket"
}

resource "aws_s3_bucket_public_access_block" "public_access" {
  bucket = aws_s3_bucket.public_bucket.id

  block_public_acls       = false  # ❌ 퍼블릭 ACL 허용
  block_public_policy     = false  # ❌ 퍼블릭 정책 허용
  ignore_public_acls      = false
  restrict_public_buckets = false
}

# ❌ 문제 2: 암호화되지 않은 EBS
resource "aws_ebs_volume" "unencrypted" {
  availability_zone = "ap-northeast-2a"
  size              = 40
  # encrypted 미설정 → 기본값 false
}

# ❌ 문제 3: 과도한 Security Group 규칙
resource "aws_security_group" "overly_permissive" {
  name        = "overly-permissive"
  description = "Too permissive"

  ingress {
    from_port   = 0
    to_port     = 0
    protocol    = "-1"
    cidr_blocks = ["0.0.0.0/0"]  # ❌ 모든 트래픽 허용
  }
}

# ❌ 문제 4: 하드코딩된 시크릿
resource "aws_db_instance" "hardcoded_password" {
  identifier     = "mydb"
  engine         = "mysql"
  instance_class = "db.t3.micro"
  username       = "admin"
  password       = "SuperSecretPassword123!"  # ❌ 하드코딩된 비밀번호
}
EOF
```

```bash
# 안전한 코드로 수정
cat > environments/dev/secure.tf << 'EOF'
# 보안 Best Practices 적용된 코드

# ✅ 안전한 S3 버킷
resource "aws_s3_bucket" "secure_bucket" {
  bucket = "tossplace-secure-bucket-${random_id.bucket.hex}"
}

resource "random_id" "bucket" {
  byte_length = 8
}

resource "aws_s3_bucket_versioning" "versioning" {
  bucket = aws_s3_bucket.secure_bucket.id
  versioning_configuration {
    status = "Enabled"
  }
}

resource "aws_s3_bucket_server_side_encryption_configuration" "encryption" {
  bucket = aws_s3_bucket.secure_bucket.id

  rule {
    apply_server_side_encryption_by_default {
      sse_algorithm     = "aws:kms"
      kms_master_key_id = aws_kms_key.s3.arn
    }
    bucket_key_enabled = true
  }
}

resource "aws_s3_bucket_public_access_block" "block_public" {
  bucket = aws_s3_bucket.secure_bucket.id

  block_public_acls       = true
  block_public_policy     = true
  ignore_public_acls      = true
  restrict_public_buckets = true
}

resource "aws_kms_key" "s3" {
  description             = "KMS key for S3 encryption"
  deletion_window_in_days = 7
  enable_key_rotation     = true
}

# ✅ 암호화된 EBS
resource "aws_ebs_volume" "encrypted" {
  availability_zone = "ap-northeast-2a"
  size              = 40
  encrypted         = true
  kms_key_id        = aws_kms_key.ebs.arn
  
  tags = {
    Name        = "encrypted-volume"
    Environment = "dev"
    Project     = "tossplace"
    ManagedBy   = "Terraform"
  }
}

resource "aws_kms_key" "ebs" {
  description             = "KMS key for EBS encryption"
  deletion_window_in_days = 7
  enable_key_rotation     = true
}

# ✅ 최소 권한 Security Group
resource "aws_security_group" "restricted" {
  name        = "restricted-access"
  description = "Restricted access"
  vpc_id      = module.vpc.vpc_id

  # 특정 포트만 허용
  ingress {
    description = "HTTPS from VPC"
    from_port   = 443
    to_port     = 443
    protocol    = "tcp"
    cidr_blocks = [module.vpc.vpc_cidr]  # VPC 내부만 허용
  }

  egress {
    description = "Allow outbound"
    from_port   = 0
    to_port     = 0
    protocol    = "-1"
    cidr_blocks = ["0.0.0.0/0"]
  }

  tags = {
    Name        = "restricted-sg"
    Environment = "dev"
    Project     = "tossplace"
    ManagedBy   = "Terraform"
  }
}

# ✅ 시크릿은 변수 또는 Secrets Manager 사용
variable "db_password" {
  description = "Database password"
  type        = string
  sensitive   = true  # 로그에 출력되지 않음
}

resource "aws_db_instance" "secure_db" {
  identifier     = "tossplace-db"
  engine         = "mysql"
  engine_version = "8.0"
  instance_class = "db.t3.micro"
  
  username = "admin"
  password = var.db_password  # 변수로 전달
  
  storage_encrypted = true
  kms_key_id        = aws_kms_key.rds.arn
  
  backup_retention_period = 7
  multi_az                = true
  deletion_protection     = true
  
  vpc_security_group_ids = [aws_security_group.restricted.id]
  
  tags = {
    Name        = "tossplace-db"
    Environment = "dev"
    Project     = "tossplace"

  }
```

### 실습 4: Pre-commit Hooks 설정 (20분)

**목표**: 커밋 전 자동 검증 구성

```bash
# Pre-commit 설정 파일
cat > .pre-commit-config.yaml << 'EOF'
repos:
  # Terraform 포맷팅 및 검증
  - repo: https://github.com/antonbabenko/pre-commit-terraform
    rev: v1.86.0
    hooks:
      - id: terraform_fmt
        name: Terraform Format
        
      - id: terraform_validate
        name: Terraform Validate
        
      - id: terraform_tflint
        name: TFLint
        args:
          - --args=--config=__GIT_WORKING_DIR__/.tflint.hcl
          
      - id: terraform_docs
        name: Terraform Docs
        args:
          - --args=--config=.terraform-docs.yml
          
      - id: terraform_checkov
        name: Checkov Security Scan
        args:
          - --args=--quiet
          - --args=--compact

  # 일반 pre-commit hooks
  - repo: https://github.com/pre-commit/pre-commit-hooks
    rev: v4.5.0
    hooks:
      - id: trailing-whitespace
      - id: end-of-file-fixer
      - id: check-yaml
      - id: check-json
      - id: check-merge-conflict
      - id: detect-private-key
        name: Detect Private Keys
      - id: detect-aws-credentials
        name: Detect AWS Credentials
        
  # Secrets 탐지
  - repo: https://github.com/Yelp/detect-secrets
    rev: v1.4.0
    hooks:
      - id: detect-secrets
        name: Detect Secrets
        args: ['--baseline', '.secrets.baseline']
EOF

cat > .terraform-docs.yml << 'EOF'

  {{ .Providers }}
```

### 실습 5: 태깅 전략 구현 (15분)

**목표**: 비용 추적 및 자동화를 위한 태그 표준화

```bash
# 태깅 모듈 생성
cat > modules/tags/main.tf << 'EOF'
# 중앙집중식 태그 관리 모듈

locals {
  # 필수 태그
  required_tags = {
    Environment = var.environment
    Project     = var.project_name
    ManagedBy   = "Terraform"
    Team        = var.team
    CostCenter  = var.cost_center
  }
  
  # 자동화 태그
  automation_tags = {
    TerraformWorkspace = terraform.workspace
    LastModified       = timestamp()
    Repository         = var.repository_url
  }
  
  # 모든 태그 병합
  all_tags = merge(
    local.required_tags,
    local.automation_tags,
    var.additional_tags
  )
}
EOF

cat > modules/tags/variables.tf << 'EOF'
variable "environment" {
  description = "Environment name"
  type        = string
  
  validation {
    condition     = contains(["dev", "staging", "prod"], var.environment)
    error_message = "Environment must be dev, staging, or prod."
  }
}

variable "project_name" {
  description = "Project name"
  type        = string
}

variable "team" {
  description = "Team responsible for the resource"
  type        = string
}

variable "cost_center" {
  description = "Cost center for billing"
  type        = string
}

variable "repository_url" {
  description = "Git repository URL"
  type        = string
  default     = "https://github.com/tossplace/infrastructure"
}

variable "additional_tags" {
  description = "Additional custom tags"
  type        = map(string)
  default     = {}
}
EOF

cat > modules/tags/outputs.tf << 'EOF'
output "tags" {
  description = "All tags to apply to resources"
  value       = local.all_tags
}

output "required_tags" {
  description = "Required tags only"
  value       = local.required_tags
}
EOF

# 사용 예시
cat > environments/dev/main.tf << 'EOF'
module "tags" {
  source = "../../modules/tags"
  
  environment    = "dev"
  project_name   = "tossplace"
  team           = "Platform"
  cost_center    = "Engineering"
  repository_url = "https://github.com/tossplace/infrastructure"
  
  additional_tags = {
    Application = "payment-gateway"
  }
}

# 모든 리소스에 태그 적용
resource "aws_instance" "example" {
  ami           = "ami-0c55b159cbfafe1f0"
  instance_type = "t3.micro"
  
  tags = merge(module.tags.tags, {
    Name = "example-instance"
  })
}
EOF
```

## 📊 Part 3: 장애 시나리오 실습 (20분)

### 시나리오 1: CI/CD에서 보안 취약점 발견

```bash
# GitHub Actions에서 Checkov 실패 시
# .github/workflows/terraform.yml

name: Terraform Security Check

on:
  pull_request:
    paths:
      - 'terraform/**'

jobs:
  security:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      
      - name: Run Checkov
        uses: bridgecrewio/checkov-action@master
        with:
          directory: terraform/
          framework: terraform
          soft_fail: false  # 실패 시 PR 블록
          output_format: github_failed_only
          
      # 실패 시 자동 코멘트
      - name: Comment on PR
        if: failure()
        uses: actions/github-script@v6
        with:
          script: |
            github.rest.issues.createComment({
              issue_number: context.issue.number,
              owner: context.repo.owner,
              repo: context.repo.repo,
              body: '🚨 Security scan failed! Please fix the issues before merging.'
            })
```

### 시나리오 2: 태그 누락으로 비용 추적 불가

```bash
# AWS Config 규칙으로 태그 강제
cat > global/config/required_tags.tf << 'EOF'
resource "aws_config_config_rule" "required_tags" {
  name = "required-tags"

  source {
    owner             = "AWS"
    source_identifier = "REQUIRED_TAGS"
  }

  input_parameters = jsonencode({
    tag1Key   = "Environment"
    tag2Key   = "Project"
    tag3Key   = "ManagedBy"
    tag4Key   = "Team"
    tag5Key   = "CostCenter"
  })

  scope {
    compliance_resource_types = [
      "AWS::EC2::Instance",
      "AWS::RDS::DBInstance",
      "AWS::S3::Bucket",
      "AWS::EKS::Cluster"
    ]
  }
}

# 비준수 리소스 자동 알림
resource "aws_cloudwatch_event_rule" "compliance_change" {
  name        = "detect-noncompliant-resources"
  description = "Detect resources without required tags"

  event_pattern = jsonencode({
    source      = ["aws.config"]
    detail-type = ["Config Rules Compliance Change"]
    detail = {
      newEvaluationResult = {
        complianceType = ["NON_COMPLIANT"]
      }
    }
  })
}
EOF
```
